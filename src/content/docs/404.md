The page you're looking for doesn't exist or may have moved.

Return to the <a href="/" data-product-home>docs home</a> or use the search above to find what you're looking for.

<script>
  // Point the "docs home" link to the root of the current product.
  // Non-core products live under a path prefix (e.g. /my-product/).
  // Core and versioned Core paths (e.g. /v0-61/...) resolve to /.
  (function () {
    var m = location.pathname.match(/^(\/[^/]+\/)/);
    if (m && !/^\/v\d+-\d+\/$/.test(m[1])) {
      var link = document.querySelector('a[data-product-home]');
      if (link) link.setAttribute('href', m[1]);
    }
  })();

  // On versioned URLs, show a hint suggesting the page may exist in the latest version.
  (function () {
    var vm = location.pathname.match(/^(\/(?:[^/]+\/)*)v\d+-\d+\//);
    if (vm) {
      var latestHref = vm[1] || '/';
      var content = document.querySelector('.sl-markdown-content');
      if (content) {
        var hint = document.createElement('p');
        hint.innerHTML = 'You\'re viewing an older version of the docs. This page may exist in the <a href="' + latestHref + '">latest version</a>.';
        content.appendChild(hint);
      }
    }
  })();

  // When a 404 occurs, fetch the appropriate 404 page for the current product/version
  // (which has the correct sidebar) and swap it in so the user sees the right navigation.
  (async function () {
    var path = window.location.pathname;

    // Don't try to fetch a 404 page for a 404 path — would loop to e.g. /404/404/.
    if (path.endsWith('/404/')) return;

    var fetch404Url;

    // Case 1: versioned URL (e.g. /core/v0-62/some-doc/ or /v0-61/some-doc/)
    // Fetch the versioned 404 page which has the correct sidebar for that version.
    var vm = path.match(/^(\/(?:[^/]+\/)*)v(\d+-\d+)\//);
    if (vm) {
      fetch404Url = vm[1] + 'v' + vm[2] + '/404/';
    } else {
      // Case 2: non-versioned product URL (e.g. /core/some-doc/)
      // Fetch the product's own 404 page which has the correct product sidebar.
      var pm = path.match(/^(\/[^/]+\/)/);
      if (pm && !/^\/v\d+-\d+\/$/.test(pm[1])) {
        fetch404Url = pm[1] + '404/';
      }
    }

    if (!fetch404Url) return;

    try {
      var res = await fetch(fetch404Url);
      if (!res.ok) return;

      var html = await res.text();
      var parser = new DOMParser();
      var doc = parser.parseFromString(html, 'text/html');

      // Swap the sidebar navigation items.
      var currentList = document.querySelector('#starlight__sidebar ul.top-level');
      var fetchedList = doc.querySelector('#starlight__sidebar ul.top-level');
      if (currentList && fetchedList) {
        currentList.replaceWith(fetchedList);
      }

      // Update the product dropdown to show the correct product as selected.
      // Swap innerHTML of the button and menu rather than replacing the elements
      // themselves — this preserves the plugin's event listeners.
      var currentButton = document.getElementById('starlight-sidebar-topics-dropdown-button');
      var fetchedButton = doc.getElementById('starlight-sidebar-topics-dropdown-button');
      if (currentButton && fetchedButton) {
        currentButton.innerHTML = fetchedButton.innerHTML;
      }

      var currentMenu = document.getElementById('starlight-sidebar-topics-dropdown-menu');
      var fetchedMenu = doc.getElementById('starlight-sidebar-topics-dropdown-menu');
      if (currentMenu && fetchedMenu) {
        currentMenu.innerHTML = fetchedMenu.innerHTML;

        // Hide versioned topics from the new menu items (mirrors Sidebar.astro logic).
        currentMenu.querySelectorAll('a').forEach(function (a) {
          if (/\/v\d+-\d+\//.test(a.getAttribute('href') || '')) {
            a.style.display = 'none';
          }
        });
      }
    } catch (e) {
      // Silently fail — keep the existing sidebar.
    }
  })();
</script>
