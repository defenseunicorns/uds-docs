The page you're looking for doesn't exist or may have moved.

If you arrived here while switching versions, use the **Version** selector in the header to navigate, or return to the [docs home](/).

<script>
  // Redirect versioned 404s to the correct versioned 404 page so the user sees
  // the right sidebar. Handles both root products (e.g. /v0-61/missing) and
  // nested products (e.g. /fleet/v1-2/missing).
  // Non-versioned product 404s (e.g. /fleet/nonexistent) stay on this page,
  // which is correct — Astro serves this 404.md for all unmatched routes.
  (function () {
    // Already on a 404 page — don't redirect again (prevents infinite loops).
    if (/\/404\/?$/.test(location.pathname)) return;
    // Try nested product path first: /product/v1-2/...
    var m = location.pathname.match(/^(\/[^/]+\/)(v\d+-\d+)\//);
    if (m) { location.replace(m[1] + m[2] + '/404/'); return; }
    // Then root-level version: /v0-61/...
    m = location.pathname.match(/^\/(v\d+-\d+)\//);
    if (m) location.replace('/' + m[1] + '/404/');
  })();
</script>
